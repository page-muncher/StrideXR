cmake_minimum_required(VERSION 3.21)
project(StrideXR VERSION 1.6.3 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_GENERATOR_PLATFORM MATCHES "^(Win32|x64)$")
    message(FATAL_ERROR "Build with Visual Studio for Windows x64 or Win32.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX XR_NO_PROTOTYPES)
add_compile_options(/W4 /permissive- /wd4351 /utf-8)
add_library(StrideXR SHARED src/layer.cpp src/audiobook_overlay.cpp)
target_include_directories(StrideXR PRIVATE third_party)
target_link_libraries(StrideXR PRIVATE d3d11 gdi32 user32)
if(CMAKE_SIZEOF_VOID_P EQUAL 4)
    target_compile_definitions(StrideXR PRIVATE STRIDEXR_X86)
    # The loader looks up the undecorated negotiation entry point by name.
    target_link_options(StrideXR PRIVATE "/EXPORT:xrNegotiateLoaderApiLayerInterface=_xrNegotiateLoaderApiLayerInterface@12")
else()
    target_link_options(StrideXR PRIVATE "/EXPORT:xrNegotiateLoaderApiLayerInterface")
    add_library(StrideXRAudio SHARED src/audiobook_audio.cpp)
    target_include_directories(StrideXRAudio PRIVATE third_party)
    target_link_libraries(StrideXRAudio PRIVATE xaudio2)
    add_executable(StrideXRCalibrate src/calibrate.cpp src/test_scene.cpp src/audiobook_overlay.cpp)
    target_include_directories(StrideXRCalibrate PRIVATE third_party)
    target_link_libraries(StrideXRCalibrate PRIVATE d3d11 dxgi d3dcompiler gdi32 user32)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(StrideXRTest tests/integration.cpp)
    add_library(StrideXRTestRuntime SHARED tests/runtime.cpp)
    target_include_directories(StrideXRTest PRIVATE third_party)
    target_link_libraries(StrideXRTest PRIVATE d3d11)
    target_include_directories(StrideXRTestRuntime PRIVATE third_party)
    target_link_libraries(StrideXRTestRuntime PRIVATE dxgi)
    if(CMAKE_SIZEOF_VOID_P EQUAL 4)
        target_link_options(StrideXRTestRuntime PRIVATE "/EXPORT:xrNegotiateLoaderRuntimeInterface=_xrNegotiateLoaderRuntimeInterface@8")
    else()
        target_link_options(StrideXRTestRuntime PRIVATE "/EXPORT:xrNegotiateLoaderRuntimeInterface")
    endif()
endif()
